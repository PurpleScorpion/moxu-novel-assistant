cmake_minimum_required(VERSION 3.21)
project(Moxu VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets Sql)
add_library(book src/book.cpp)
target_include_directories(book PUBLIC src)
target_link_libraries(book PUBLIC Qt6::Sql)
add_executable(Moxu WIN32 src/main.cpp)
target_link_libraries(Moxu PRIVATE Qt6::Widgets book)
find_file(QT_CHINESE_CATALOG qtbase_zh_CN.qm HINTS "${Qt6_DIR}/../../../translations" REQUIRED)
get_filename_component(QT_CATALOG_DIR "${QT_CHINESE_CATALOG}" DIRECTORY)
qt_add_resources(Moxu chinese_ui PREFIX "/lang" BASE "${QT_CATALOG_DIR}" FILES "${QT_CHINESE_CATALOG}")
enable_testing()
add_executable(book_test tests/book_test.cpp)
target_link_libraries(book_test PRIVATE book)
add_test(NAME persistence COMMAND book_test)
find_package(Qt6 REQUIRED COMPONENTS Test)
add_executable(writer_test tests/writer_test.cpp)
target_link_libraries(writer_test PRIVATE book Qt6::Widgets Qt6::Test)
add_test(NAME writer COMMAND writer_test)
