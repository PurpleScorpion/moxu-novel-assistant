cmake_minimum_required(VERSION 3.21)
project(Moxu VERSION 0.27.2 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets Sql Network)
include(FetchContent)
if(POLICY CMP0135)
    cmake_policy(SET CMP0135 NEW)
endif()
set(YAML_CPP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(YAML_CPP_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(YAML_CPP_BUILD_CONTRIB OFF CACHE BOOL "" FORCE)
set(YAML_BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(yaml-cpp
    URL https://codeload.github.com/jbeder/yaml-cpp/tar.gz/refs/tags/0.8.0
    URL_HASH SHA256=fbe74bbdcee21d656715688706da3c8becfd946d92cd44705cc6098bb23b3a16)
function(fetch_skill_yaml)
    set(CMAKE_POLICY_VERSION_MINIMUM 3.10)
    FetchContent_MakeAvailable(yaml-cpp)
    set(SKILL_YAML_LICENSE "${yaml-cpp_SOURCE_DIR}/LICENSE" PARENT_SCOPE)
endfunction()
fetch_skill_yaml()
add_library(book src/book.cpp src/skill_file.cpp)
target_include_directories(book PUBLIC src)
target_link_libraries(book PUBLIC Qt6::Sql PRIVATE yaml-cpp::yaml-cpp)
add_executable(Moxu WIN32 src/main.cpp)
if(WIN32)
    enable_language(RC)
    target_sources(Moxu PRIVATE assets/moxu.rc)
    target_include_directories(Moxu PRIVATE assets)
    set_property(SOURCE assets/moxu.rc PROPERTY OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/moxu.ico")
endif()
qt_add_resources(Moxu app_icon PREFIX "/" FILES assets/moxu.png)
target_link_libraries(Moxu PRIVATE Qt6::Widgets Qt6::Network book)
add_custom_command(TARGET Moxu POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different
    "${SKILL_YAML_LICENSE}" "$<TARGET_FILE_DIR:Moxu>/yaml-cpp-LICENSE.txt")
find_file(QT_CHINESE_CATALOG qtbase_zh_CN.qm HINTS "${Qt6_DIR}/../../../translations" REQUIRED)
get_filename_component(QT_CATALOG_DIR "${QT_CHINESE_CATALOG}" DIRECTORY)
qt_add_resources(Moxu chinese_ui PREFIX "/lang" BASE "${QT_CATALOG_DIR}" FILES "${QT_CHINESE_CATALOG}")
enable_testing()
add_executable(book_test tests/book_test.cpp)
target_link_libraries(book_test PRIVATE book)
add_test(NAME persistence COMMAND book_test)
find_package(Qt6 REQUIRED COMPONENTS Test)
add_executable(writer_test tests/writer_test.cpp)
target_link_libraries(writer_test PRIVATE book Qt6::Widgets Qt6::Network Qt6::Test)
add_test(NAME writer COMMAND writer_test)
add_executable(writing_test tests/writing_test.cpp)
target_link_libraries(writing_test PRIVATE book Qt6::Widgets Qt6::Network Qt6::Test)
add_test(NAME writing_skills COMMAND writing_test)
add_executable(skill_file_test tests/skill_file_test.cpp)
target_link_libraries(skill_file_test PRIVATE book Qt6::Widgets Qt6::Network)
add_test(NAME skill_file_exchange COMMAND skill_file_test)
add_executable(focus_test tests/focus_test.cpp)
target_link_libraries(focus_test PRIVATE book Qt6::Widgets Qt6::Network Qt6::Test)
add_test(NAME focus_without_dialogs COMMAND focus_test)
if(WIN32)
    foreach(target Moxu writer_test writing_test skill_file_test focus_test)
        target_link_libraries(${target} PRIVATE advapi32)
    endforeach()
endif()
